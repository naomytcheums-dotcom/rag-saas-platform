# a separate file from requirements.txt (the RAG pipeline's own deps) so
# `pip install -r requirements.txt` still works standalone for anyone only
# touching src/ or dashboard/ -- installing both files together is what a
# full `uvicorn api.main:app` run actually needs.
#
# Exact versions frozen from the environment this was built and tested
# against (Python 3.13, Windows) -- same reasoning as requirements.txt's
# own header: relax the pin and re-run `pytest tests/test_auth_*.py`
# before trusting the result. Passlib is deliberately NOT used for
# password hashing -- see api/security/hashing.py's docstring: it's
# unmaintained and its own backend self-test crashes against current
# bcrypt releases.

# -- Monitoring (audit finding 22) --------------------------------------------
# prometheus-client's multiprocess mode (api/monitoring.py) is the
# actual reason this is a real dependency and not hand-rolled like
# api/security/rate_limit.py or password_similarity.py -- correctly
# aggregating histograms across several worker PROCESSES (mmap'd
# per-worker files, cleaned up as workers come and go) is real,
# easy-to-get-wrong complexity a battle-tested library exists
# specifically to solve, not "a few dozen lines to read in a few
# minutes." gunicorn is Unix-only (uses fcntl -- won't run on Windows,
# though `pip install` itself succeeds there) and only needed to run
# gunicorn.conf.py in actual production; local dev keeps using
# `uvicorn api.main:app --reload` either way.
prometheus-client==0.26.0
gunicorn==23.0.0
