# -- Monitoring (audit finding 22) --------------------------------------------
# prometheus-client's multiprocess mode (api/monitoring.py) is the
# actual reason this is a real dependency and not hand-rolled like
# api/security/rate_limit.py or password_similarity.py -- correctly
# aggregating histograms across several worker PROCESSES (mmap'd
# per-worker files, cleaned up as workers come and go) is real,
# easy-to-get-wrong complexity a battle-tested library exists
# specifically to solve, not "a few dozen lines to read in a few
# minutes." gunicorn is Unix-only (uses fcntl -- won't run on Windows,
# though `pip install` itself succeeds there) and only needed to run
# gunicorn.conf.py in actual production; local dev keeps using
# `uvicorn api.main:app --reload` either way.
prometheus-client==0.26.0
gunicorn==23.0.0

# -- RBAC / policy engine (Etape 1.2.7) ---------------------------------------
# PyPI publishes this project under TWO distribution names -- "casbin"
# (legacy, frozen at an old 1.x version line) and "pycasbin" (current,
# what casbin-async-sqlalchemy-adapter itself actually depends on) --
# both install into the same importable `casbin` package, so installing
# both is fragile (whichever installs last silently wins). pycasbin is
# the one actually maintained; api/security/rbac.py's AsyncEnforcer usage
# requires it (pycasbin>=2.x), not the legacy "casbin" line.
pycasbin==2.8.0
# Async, not the sync casbin-sqlalchemy-adapter the original ask named --
# this codebase's entire DB layer (api/database.py) is async SQLAlchemy;
# the sync adapter would need its own separate sync engine talking to the
# same Postgres instance outside SQLAlchemy's async session management,
# the same category of footgun documented elsewhere in this codebase
# (MissingGreenlet, sync lazy-loads under an async session).
casbin-async-sqlalchemy-adapter==1.17.0
